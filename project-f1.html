Project 01 · Data Science & BI

F1 Race Telemetry & Strategy Pipeline

An interactive telemetry analysis system translating sub-second vehicle kinematics into visual competitive insights.

Interactive Analytics App

Select a circuit and choose top-tier constructor drivers to visualize cornering velocity metrics alongside spatial tracking lines.

Data Architecture

Raw asynchronous data signals sample telemetry updates at ~20Hz. To protect dashboard application rendering, a custom Python architecture maps physical vehicle metrics, handles cache limits, and downsamples spatial dimensions uniformly to a manageable 10Hz sequence.

View Data Engineering Pipeline Script 
import os
import pandas as pd
import fastf1

cache_dir = './f1_cache'
os.makedirs(cache_dir, exist_ok=True)
fastf1.Cache.enable_cache(cache_dir)

def build_master_telemetry(year, races, drivers):
    master_df = []
    for race in races:
        session = fastf1.get_session(year, race, 'Q')
        session.load(telemetry=True, laps=True, weather=False)
        
        for driver in drivers:
            driver_laps = session.laps.pick_drivers(driver)
            if driver_laps.empty: continue
            
            fastest_lap = driver_laps.pick_fastest()
            tel = fastest_lap.get_telemetry().add_distance()
            tel = tel.iloc[::2].copy() # Downsample to 10Hz
            
            tel['Race'] = race
            tel['Year'] = year
            tel['Driver'] = driver
            tel['Team'] = fastest_lap['Team']
            
            keep_cols = ['Year', 'Race', 'Distance', 'Speed', 'Throttle', 'Brake', 
                         'RPM', 'nGear', 'X', 'Y', 'Driver', 'Team']
            df_cleaned = tel[keep_cols].copy()
            df_cleaned.rename(columns={'nGear': 'Gear'}, inplace=True)
            master_df.append(df_cleaned)
            
    final_df = pd.concat(master_df, ignore_index=True)
    final_df.to_csv(f"{year}_master_f1_telemetry.csv", index=False)
    return final_df
